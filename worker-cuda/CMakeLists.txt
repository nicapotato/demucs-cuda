cmake_minimum_required(VERSION 3.24)
project(demucs_cuda_worker LANGUAGES C CXX)

set(DC_BACKEND "cuda" CACHE STRING "Kernel backend: cuda (NVIDIA) or cpu (portable reference)")
set_property(CACHE DC_BACKEND PROPERTY STRINGS cuda cpu)
# SASS for every listed arch, plus PTX for the last one (forward compatibility).
set(DC_CUDA_ARCHS "75;80;86;89;120" CACHE STRING "CUDA compute capabilities to build")
option(DC_BUILD_TESTS "Build kernel / golden comparison tests" ON)
option(DC_HALF_FLOAT "cpu backend only: keep GEMM operands in fp32 (numerics experiments)" OFF)
if(DC_HALF_FLOAT)
  if(NOT DC_BACKEND STREQUAL "cpu")
    message(FATAL_ERROR "DC_HALF_FLOAT requires DC_BACKEND=cpu")
  endif()
  add_compile_definitions(DC_HALF_FLOAT)
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(MSVC)
  # static CRT: no VC++ redistributable needed on the target machine
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
endif()

set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/src)
set(TP ${CMAKE_CURRENT_SOURCE_DIR}/third_party)

add_library(dc_core STATIC
  ${SRC}/util.c ${SRC}/config.c ${SRC}/tensor.c ${SRC}/weights.c ${SRC}/ops.c
  ${SRC}/stft.c ${SRC}/layers.c ${SRC}/transformer.c ${SRC}/htdemucs.c ${SRC}/apply.c)
target_include_directories(dc_core PUBLIC ${SRC})
if(NOT MSVC)
  target_link_libraries(dc_core PUBLIC m)
  target_compile_options(dc_core PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()

if(DC_BACKEND STREQUAL "cuda")
  set(_archs "")
  list(LENGTH DC_CUDA_ARCHS _n)
  math(EXPR _last "${_n} - 1")
  foreach(_i RANGE ${_last})
    list(GET DC_CUDA_ARCHS ${_i} _a)
    if(_i EQUAL _last)
      list(APPEND _archs "${_a}")
    else()
      list(APPEND _archs "${_a}-real")
    endif()
  endforeach()
  set(CMAKE_CUDA_ARCHITECTURES "${_archs}")
  set(CMAKE_CUDA_RUNTIME_LIBRARY Static)
  set(CMAKE_CUDA_STANDARD 17)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)

  set(DC_BACKEND_SRC ${SRC}/backend_cuda.cu ${SRC}/gemm.cu ${SRC}/norm.cu ${SRC}/attn.cu
                     ${SRC}/fft.cu)
  add_library(dc_backend STATIC ${DC_BACKEND_SRC})
  target_include_directories(dc_backend PRIVATE ${SRC} ${TP}/cutlass/include)
  target_compile_options(dc_backend PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr -lineinfo>
    $<$<AND:$<COMPILE_LANGUAGE:CUDA>,$<CXX_COMPILER_ID:MSVC>>:-Xcompiler=/bigobj -Xcompiler=/Zc:__cplusplus>)
  if(MSVC AND CMAKE_VERSION VERSION_LESS 3.27)
    # MSVC_RUNTIME_LIBRARY only reaches nvcc's host compiler from CMake 3.27
    target_compile_options(dc_backend PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/MT>)
  endif()
  set_target_properties(dc_backend PROPERTIES CUDA_RUNTIME_LIBRARY Static)
  target_link_libraries(dc_backend PUBLIC CUDA::cudart_static)
  set(DC_EXE demucs_cuda_worker)
elseif(DC_BACKEND STREQUAL "cpu")
  add_library(dc_backend STATIC ${SRC}/backend_cpu.c)
  target_include_directories(dc_backend PRIVATE ${SRC})
  if(APPLE)
    target_compile_definitions(dc_backend PRIVATE DC_USE_ACCELERATE ACCELERATE_NEW_LAPACK)
    target_link_libraries(dc_backend PUBLIC "-framework Accelerate")
  endif()
  if(NOT MSVC)
    target_link_libraries(dc_backend PUBLIC m)
  endif()
  set(DC_EXE demucs_cpu_worker)
else()
  message(FATAL_ERROR "DC_BACKEND must be cuda or cpu")
endif()

# ---- lame (mp3 output) ----
find_package(mp3lame CONFIG QUIET)
if(TARGET mp3lame::mp3lame)
  set(DC_LAME mp3lame::mp3lame)
else()
  find_path(LAME_INCLUDE_DIR lame/lame.h HINTS /opt/homebrew/include /usr/local/include)
  find_library(LAME_LIBRARY NAMES libmp3lame.a mp3lame libmp3lame libmp3lame-static
               HINTS /opt/homebrew/lib /usr/local/lib)
  if(NOT LAME_INCLUDE_DIR OR NOT LAME_LIBRARY)
    message(FATAL_ERROR "lame not found (vcpkg: mp3lame, brew: lame, apt: libmp3lame-dev)")
  endif()
  add_library(dc_lame INTERFACE)
  target_include_directories(dc_lame INTERFACE ${LAME_INCLUDE_DIR})
  target_link_libraries(dc_lame INTERFACE ${LAME_LIBRARY})
  # static lame built with mpg123 decoding (Homebrew, vcpkg) needs it at link time
  if(LAME_LIBRARY MATCHES "\\.(a|lib)$")
    find_library(MPG123_LIBRARY NAMES libmpg123.a mpg123 libmpg123 HINTS /opt/homebrew/lib /usr/local/lib)
    if(MPG123_LIBRARY)
      target_link_libraries(dc_lame INTERFACE ${MPG123_LIBRARY})
      if(WIN32)
        target_link_libraries(dc_lame INTERFACE shlwapi)
      endif()
    endif()
  endif()
  set(DC_LAME dc_lame)
endif()

add_library(dc_audio STATIC ${SRC}/audio.c $<$<PLATFORM_ID:Windows>:${SRC}/audio_mf.cpp>)
target_include_directories(dc_audio PRIVATE ${TP}/dr_libs ${TP}/stb)
target_link_libraries(dc_audio PUBLIC dc_core ${DC_LAME})
if(WIN32)
  target_link_libraries(dc_audio PUBLIC mfplat mfreadwrite mfuuid ole32)
endif()

add_executable(${DC_EXE} ${SRC}/main.c)
target_link_libraries(${DC_EXE} PRIVATE dc_audio dc_core dc_backend)
if(DC_BACKEND STREQUAL "cuda")
  set_target_properties(${DC_EXE} PROPERTIES LINKER_LANGUAGE CUDA CUDA_RUNTIME_LIBRARY Static)
endif()
install(TARGETS ${DC_EXE} RUNTIME DESTINATION .)

if(DC_BUILD_TESTS)
  foreach(t test_kernels test_stft test_compare)
    add_executable(${t} ${CMAKE_CURRENT_SOURCE_DIR}/tests/${t}.c)
    target_link_libraries(${t} PRIVATE dc_core dc_backend)
    if(DC_BACKEND STREQUAL "cuda")
      set_target_properties(${t} PROPERTIES LINKER_LANGUAGE CUDA CUDA_RUNTIME_LIBRARY Static)
    endif()
  endforeach()
  enable_testing()
  add_test(NAME kernels COMMAND test_kernels)
  add_test(NAME stft COMMAND test_stft)
endif()
